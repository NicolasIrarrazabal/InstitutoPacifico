package com.proyecto.ms_empresas;

import io.swagger.v3.oas.models.OpenAPI;
import io.swagger.v3.oas.models.info.Info;
import io.swagger.v3.oas.models.servers.Server;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.util.List;

@Configuration
public class OpenApiConfig {

    @Bean
    public OpenAPI customOpenAPI() {
        return new OpenAPI()
                .info(new Info()
                        .title("ms-empresas")
                        .description("Microservicio de gestión de empresas con convenio - Instituto Pacífico. " +
                                "Gestiona empresas y convenios para prácticas profesionales (R5).")
                        .version("1.0.0"))
                .servers(List.of(new Server().url("http://localhost:8089").description("Local")));
    }
}
